from datetime import datetime

# Define the format of the timestamp
time_format = "%a %d %b %Y %H:%M:%S %z"

# Read number of test cases
n = int(input())

for _ in range(n):
    # Read two timestamps
    t1 = input().strip()
    t2 = input().strip()
    
    # Parse them into datetime objects
    dt1 = datetime.strptime(t1, time_format)
    dt2 = datetime.strptime(t2, time_format)
    
    # Compute absolute difference in seconds
    diff = abs(int((dt1 - dt2).total_seconds()))
    print(diff)
