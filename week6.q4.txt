class EvenStream:
    def __init__(self):
        self.current = 0

    def get_next(self):
        self.current += 2
        return self.current - 2


class OddStream:
    def __init__(self):
        self.current = 1

    def get_next(self):
        self.current += 2
        return self.current - 2


def print_from_stream(n, stream=None):
    # Default to EvenStream if no stream is provided
    if stream is None:
        stream = EvenStream()
    for _ in range(n):
        print(stream.get_next())


# The locked template will call this function based on input
if __name__ == "__main__":
    q = int(input())
    for _ in range(q):
        stream_name, n = input().split()
        n = int(n)
        if stream_name == "even":
            print_from_stream(n)
        else:
            print_from_stream(n, OddStream())
