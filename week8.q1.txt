import java.util.Scanner;

public class Solution {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int q = sc.nextInt();  // number of queries

        for (int t = 0; t < q; t++) {
            int a = sc.nextInt();
            int b = sc.nextInt();
            int n = sc.nextInt();

            int current = a;
            for (int i = 0; i < n; i++) {
                current += (int) Math.pow(2, i) * b;
                System.out.print(current + " ");
            }
            System.out.println();
        }

        sc.close();
    }
}
